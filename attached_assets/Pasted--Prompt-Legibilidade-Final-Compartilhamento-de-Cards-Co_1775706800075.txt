## BLOCO 3 — COMPARTILHAMENTO DE CARDS: CORRIGIR TUDO

### Problema observado

Ao compartilhar, os cards aparecem completamente em branco ou preto — sem o conteúdo escrito. Isso acontece porque:

1. A fonte Cormorant Garamond não está carregada no momento em que o `html2canvas` (ou lib equivalente) tira o screenshot
1. Elementos com `position: absolute` ou `transform` às vezes não são capturados corretamente

### Solução completa

Substituir a abordagem atual de captura de screenshot por **geração de card em canvas puro**, sem depender de renderização HTML. Isso garante que funcione em 100% dos casos, independente de carregamento de fontes.

#### Implementação

Criar um arquivo `src/utils/shareCard.ts` com a seguinte lógica:

```typescript
interface ShareCardOptions {
  type: "post" | "citation_light" | "citation_dark" | "reader_identity";
  excerpt?: string;
  commentary?: string;
  bookTitle?: string;
  author?: string;
  username?: string;
  reactionCount?: number;
  topReactions?: Array<{ emoji: string; count: number }>;
  archetype?: string;
  readingSignature?: string;
  format?: "story" | "feed"; // 9:16 ou 1:1
}

export async function generateShareCard(options: ShareCardOptions): Promise<Blob> {
  const isStory = options.format !== "feed";
  const width = isStory ? 1080 : 1080;
  const height = isStory ? 1920 : 1080;

  const canvas = document.createElement("canvas");
  canvas.width = width;
  canvas.height = height;
  const ctx = canvas.getContext("2d")!;

  // Carregar fontes antes de desenhar
  await document.fonts.load("italic 300 48px 'Cormorant Garamond'");
  await document.fonts.load("400 28px 'Jost'");

  if (options.type === "citation_light") {
    await drawCitationLight(ctx, width, height, options);
  } else if (options.type === "citation_dark") {
    await drawCitationDark(ctx, width, height, options);
  } else if (options.type === "reader_identity") {
    await drawReaderIdentity(ctx, width, height, options);
  } else {
    await drawPostCard(ctx, width, height, options);
  }

  return new Promise((resolve) => {
    canvas.toBlob((blob) => resolve(blob!), "image/png", 0.95);
  });
}

// ─── CITAÇÃO LIGHT ────────────────────────────────────────────

async function drawCitationLight(
  ctx: CanvasRenderingContext2D,
  w: number,
  h: number,
  opts: ShareCardOptions
) {
  // Fundo
  ctx.fillStyle = "#FAF8F3";
  roundRect(ctx, 0, 0, w, h, 48);
  ctx.fill();

  // Textura pontilhada (dots discretos)
  ctx.fillStyle = "rgba(174, 143, 125, 0.08)";
  for (let x = 40; x < w; x += 32) {
    for (let y = 40; y < h; y += 32) {
      ctx.beginPath();
      ctx.arc(x, y, 1.5, 0, Math.PI * 2);
      ctx.fill();
    }
  }

  // Label topo
  ctx.fillStyle = "#AE8F7D";
  ctx.font = "400 26px 'Jost'";
  ctx.letterSpacing = "4px";
  fillTextWrapped(ctx, "• CITAÇÃO FAVORITA", 80, 100, w - 160, 40);

  // Logo direita
  ctx.fillStyle = "#AE8F7D";
  ctx.font = "italic 300 32px 'Cormorant Garamond'";
  ctx.textAlign = "right";
  ctx.fillText("marginalia", w - 80, 100);
  ctx.textAlign = "left";

  // Aspas decorativas
  ctx.fillStyle = "#D4CBB8";
  ctx.font = "italic 300 120px 'Cormorant Garamond'";
  ctx.fillText("\u201C", 80, 280);

  // Trecho principal
  ctx.fillStyle = "#1E1C19";
  ctx.font = "italic 300 52px 'Cormorant Garamond'";
  const excerptY = fillTextWrapped(ctx, opts.excerpt ?? "", 80, 340, w - 160, 72);

  // Livro e autor
  if (opts.bookTitle) {
    ctx.fillStyle = "#7A726A";
    ctx.font = "400 30px 'Jost'";
    ctx.fillText(`— ${opts.bookTitle}`, 80, excerptY + 60);
    if (opts.author) {
      ctx.fillStyle = "#9C948C";
      ctx.font = "400 26px 'Jost'";
      ctx.letterSpacing = "2px";
      ctx.fillText(opts.author.toUpperCase(), 80, excerptY + 104);
    }
  }

  // Linha separadora
  ctx.strokeStyle = "#D4CBB8";
  ctx.lineWidth = 1.5;
  ctx.setLineDash([8, 8]);
  ctx.beginPath();
  ctx.moveTo(80, h - 180);
  ctx.lineTo(w - 80, h - 180);
  ctx.stroke();
  ctx.setLineDash([]);

  // Rodapé: avatar + username + reações
  drawFooter(ctx, w, h, opts, "light");
}

// ─── UTILITÁRIOS ──────────────────────────────────────────────

function fillTextWrapped(
  ctx: CanvasRenderingContext2D,
  text: string,
  x: number,
  y: number,
  maxWidth: number,
  lineHeight: number
): number {
  const words = text.split(" ");
  let line = "";
  let currentY = y;

  for (const word of words) {
    const testLine = line + word + " ";
    const metrics = ctx.measureText(testLine);
    if (metrics.width > maxWidth && line !== "") {
      ctx.fillText(line.trim(), x, currentY);
      line = word + " ";
      currentY += lineHeight;
    } else {
      line = testLine;
    }
  }
  if (line.trim()) {
    ctx.fillText(line.trim(), x, currentY);
    currentY += lineHeight;
  }
  return currentY;
}

function roundRect(
  ctx: CanvasRenderingContext2D,
  x: number,
  y: number,
  w: number,
  h: number,
  r: number
) {
  ctx.beginPath();
  ctx.moveTo(x + r, y);
  ctx.lineTo(x + w - r, y);
  ctx.quadraticCurveTo(x + w, y, x + w, y + r);
  ctx.lineTo(x + w, y + h - r);
  ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
  ctx.lineTo(x + r, y + h);
  ctx.quadraticCurveTo(x, y + h, x, y + h - r);
  ctx.lineTo(x, y + r);
  ctx.quadraticCurveTo(x, y, x + r, y);
  ctx.closePath();
}

function drawFooter(
  ctx: CanvasRenderingContext2D,
  w: number,
  h: number,
  opts: ShareCardOptions,
  theme: "light" | "dark"
) {
  const isDark = theme === "dark";
  const footerY = h - 140;

  // Avatar placeholder
  ctx.fillStyle = isDark ? "#3D3830" : "#EBE6DB";
  ctx.beginPath();
  ctx.arc(112, footerY, 40, 0, Math.PI * 2);
  ctx.fill();

  // Iniciais
  ctx.fillStyle = isDark ? "#9C948C" : "#697962";
  ctx.font = "500 32px 'Jost'";
  ctx.textAlign = "center";
  const initials = (opts.username ?? "MG").slice(0, 2).toUpperCase();
  ctx.fillText(initials, 112, footerY + 11);
  ctx.textAlign = "left";

  // Username
  ctx.fillStyle = isDark ? "#E8E2D6" : "#1E1C19";
  ctx.font = "500 30px 'Jost'";
  ctx.fillText(`@${opts.username ?? "usuario"}`, 172, footerY + 4);

  // Reações
  if (opts.topReactions && opts.topReactions.length > 0) {
    let reactionX = 172;
    const reactionY = footerY + 44;
    for (const r of opts.topReactions.slice(0, 3)) {
      ctx.font = "28px serif";
      ctx.fillText(r.emoji, reactionX, reactionY);
      ctx.fillStyle = isDark ? "#9C948C" : "#7A726A";
      ctx.font = "400 26px 'Jost'";
      ctx.fillText(String(r.count), reactionX + 38, reactionY);
      reactionX += 100;
    }
  }

  // Logo
  ctx.fillStyle = isDark ? "#7A726A" : "#9C948C";
  ctx.font = "italic 300 28px 'Cormorant Garamond'";
  ctx.textAlign = "right";
  ctx.fillText("marginalia", w - 80, footerY + 4);
  ctx.textAlign = "left";
}
```

#### Conectar ao componente de compartilhamento

No componente de modal de compartilhamento (`ShareModal`, `ShareButton`, ou como estiver nomeado), substituir a lógica atual de `html2canvas` por:

```typescript
import { generateShareCard } from "@/utils/shareCard";

const handleGenerate = async () => {
  setIsGenerating(true);
  try {
    const blob = await generateShareCard({
      type: selectedStyle, // "citation_light" | "citation_dark" | "post"
      excerpt: margin.excerpt ?? undefined,
      commentary: margin.commentary ?? undefined,
      bookTitle: book?.title ?? undefined,
      author: book?.author ?? undefined,
      username: currentUser.username,
      topReactions: Object.entries(margin.reactions ?? {})
        .sort((a, b) => b[1] - a[1])
        .slice(0, 3)
        .map(([emoji, count]) => ({ emoji, count })),
      format: selectedFormat, // "story" | "feed"
    });

    // Criar URL do blob para preview
    const url = URL.createObjectURL(blob);
    setPreviewUrl(url);
  } finally {
    setIsGenerating(false);
  }
};

const handleShare = async () => {
  if (!previewBlob) return;
  const file = new File([previewBlob], "marginalia-card.png", { type: "image/png" });

  if (navigator.canShare?.({ files: [file] })) {
    await navigator.share({ files: [file], title: "Marginalia" });
  } else {
    // Fallback: download
    const a = document.createElement("a");
    a.href = URL.createObjectURL(previewBlob);
    a.download = "marginalia-card.png";
    a.click();
  }
};

const handleDownload = () => {
  if (!previewBlob) return;
  const a = document.createElement("a");
  a.href = URL.createObjectURL(previewBlob);
  a.download = "marginalia-card.png";
  a.click();
};
```

## OBSERVAÇÕES FINAIS

- A geração via Canvas elimina completamente o problema de fontes não carregadas — o `document.fonts.load()` força o carregamento antes de qualquer desenho
- O preview no modal deve atualizar automaticamente quando o usuário troca entre “Post”, “Citação Light” e “Citação Dark” — chamar `handleGenerate` sempre que `selectedStyle` ou `selectedFormat` mudar (useEffect com dependências)
- Garantir que o estado de loading (`isGenerating`) mostra um spinner no lugar do preview enquanto o canvas é gerado
- O botão BAIXAR usa download direto, o botão COMPARTILHAR usa Web Share API com fallback para download
- Testar em iOS Safari — o `navigator.canShare` funciona bem lá para arquivos de imagem
- Não remover os estilos visuais do modal (os chips de seleção de estilo e formato devem continuar funcionando normalmente)
