---

## CONTEXTO

Implementar no Marginalia (marginalia.replit.app) um sistema completo de **arquétipos de leitor** com identidade visual única por tipo, integrado ao perfil do usuário e ao sistema de cards compartilháveis.

O sistema deve parecer algo que o usuário **quer mostrar para o mundo** — identidade emocional, não apenas dado de uso.

## PARTE 2 — ALGORITMO DE DETECÇÃO

### Como calcular o arquétipo do usuário

```javascript
function calcularArquetipos(dadosUsuario) {
  // dadosUsuario contém:
  // - tiposDeMargemUsados: { insight: N, teoria: N, reacao: N, ... }
  // - totalDestaques: N
  // - totalAnotacoes: N
  // - mediaAnotacoesPorDestaque: N
  // - sessoesLeitura: [{ duracao, paginasLidas }]
  // - percentualConcluidoPorSemana: N
  // - usaEmojiIntensidade: boolean
  // - mencionaOutrosLivros: boolean
  // - usaPrimeirasPessoa: boolean (frequência de "eu" nas notas)

  const scores = {};

  // Inicializa todos com 0
  ARQUETIPOS.forEach(a => { scores[a.id] = 0; });

  const m = dadosUsuario.tiposDeMargemUsados;
  const total = Object.values(m).reduce((s, v) => s + v, 0) || 1;

  // Observador: usa insight, anota sem diálogo
  scores.observador += (m.insight / total) * 40;
  scores.observador += Math.min(dadosUsuario.mediaAnotacoesPorDestaque * 5, 30);
  scores.observador += dadosUsuario.usaPrimeiraPessoa ? 0 : 30;

  // Sensível: usa reação + citação, emojis intensos
  scores.sensivel += (m.reacao / total) * 35;
  scores.sensivel += ((m.citacaoFavorita || 0) / total) * 25;
  scores.sensivel += dadosUsuario.usaEmojiIntensidade ? 40 : 0;

  // Analítico: usa teoria + crítica + pergunta
  scores.analitico += ((m.teoria || 0) / total) * 35;
  scores.analitico += ((m.critica || 0) / total) * 30;
  scores.analitico += ((m.pergunta || 0) / total) * 35;

  // Intenso: muitos destaques por sessão, sessões longas
  const mediaDestPorSessao = dadosUsuario.totalDestaques /
    Math.max(dadosUsuario.sessoesLeitura.length, 1);
  scores.intenso += Math.min(mediaDestPorSessao * 8, 50);
  const mediaHoras = dadosUsuario.sessoesLeitura
    .reduce((s, sess) => s + sess.duracao, 0) /
    Math.max(dadosUsuario.sessoesLeitura.length, 1);
  scores.intenso += Math.min(mediaHoras * 15, 50);

  // Contemplativo: progresso lento, notas profundas
  scores.contemplativo += dadosUsuario.percentualConcluidoPorSemana < 5 ? 50 : 10;
  scores.contemplativo += Math.min(dadosUsuario.mediaAnotacoesPorDestaque * 8, 50);

  // Conector: menciona outros livros
  scores.conector += dadosUsuario.mencionaOutrosLivros ? 60 : 0;
  scores.conector += ((m.conexaoPessoal || 0) / total) * 40;

  // Interpretador: leitura simbólica
  scores.interpretador += ((m.leituraSimbolica || 0) / total) * 60;
  scores.interpretador += ((m.insight || 0) / total) * 40;

  // Imersivo: progresso rápido, poucas pauses
  scores.imersivo += dadosUsuario.percentualConcluidoPorSemana > 15 ? 60 : 10;
  const totalDestaquesPorPagina = dadosUsuario.totalDestaques /
    Math.max(dadosUsuario.totalPaginasLidas, 1);
  scores.imersivo += totalDestaquesPorPagina < 0.3 ? 40 : 0;

  // Curioso: pergunta frequente
  scores.curioso += ((m.pergunta || 0) / total) * 50;
  scores.curioso += ((m.teoria || 0) / total) * 30;
  scores.curioso += dadosUsuario.buscaInformacoesExternas ? 20 : 0;

  // Seletivo: poucos destaques, notas longas
  scores.seletivo += dadosUsuario.totalDestaques < 20 ? 40 : 0;
  scores.seletivo += dadosUsuario.mediaAnotacoesPorDestaque > 3 ? 40 : 0;
  scores.seletivo += ((m.citacaoFavorita || 0) / total) * 20;

  // Introspectivo: primeira pessoa, conexão pessoal
  scores.introspectivo += dadosUsuario.usaPrimeiraPessoa ? 50 : 0;
  scores.introspectivo += ((m.conexaoPessoal || 0) / total) * 50;

  // Estético: citação favorita dominante
  scores.estetico += ((m.citacaoFavorita || 0) / total) * 60;
  scores.estetico += dadosUsuario.comentaEstiloAutor ? 40 : 0;

  // Normaliza 0-100
  const maxScore = Math.max(...Object.values(scores), 1);
  const normalized = {};
  Object.keys(scores).forEach(k => {
    normalized[k] = Math.round((scores[k] / maxScore) * 100);
  });

  // Retorna top 3 arquétipos
  const ordenados = Object.entries(normalized)
    .sort(([,a], [,b]) => b - a)
    .slice(0, 3)
    .map(([id, score]) => ({
      arquetipo: ARQUETIPOS.find(a => a.id === id),
      score
    }));

  return ordenados;
}
```

```javascript
function getFraseCombinada(tipo1id, tipo2id) {
  const chave1 = `${tipo1id}+${tipo2id}`;
  const chave2 = `${tipo2id}+${tipo1id}`;
  return FRASES_COMBINADAS[chave1]
    || FRASES_COMBINADAS[chave2]
    || `Você ${ARQUETIPOS.find(a=>a.id===tipo1id).frase.toLowerCase().replace('você ','')} e ${ARQUETIPOS.find(a=>a.id===tipo2id).frase.toLowerCase().replace('você ','')}`;
}
```

## PARTE 4 — ESTRUTURA DO PERFIL REDESENHADO

### Ordem dos blocos na tela de Perfil:

```
1. HEADER DE IDENTIDADE
   ├── Avatar com cor do arquétipo principal
   ├── Nome + @username
   ├── Bolinha colorida com cor do arquétipo
   └── "O Sensível · Leitor intenso" — tipo principal + secundário

2. BOTÃO: "Compartilhar identidade" — gera Card Identidade

3. ASSINATURA
   └── Frase curta do usuário (editável)
       Ex: "Cada livro me deixa diferente"

4. DNA DE LEITURA
   ├── Barras de progresso para cada traço detectado
   ├── Top 4 traços com percentuais
   └── Cor da barra = cor do arquétipo correspondente

5. IMPRESSÃO DE LEITURA
   └── Bloco com frase combinada dos top 2 arquétipos
       Cormorant Garamond italic grande
       Cor do texto em tom do arquétipo principal

6. CARDS DISPONÍVEIS
   └── Grid horizontal scroll de cards para compartilhar
```

## PARTE 6 — EXPORTAÇÃO DE CARDS COMO IMAGEM

```javascript
// Usar html2canvas ou similar para exportar

import html2canvas from 'html2canvas';

async function exportarCard(elementId, nomeArquivo) {
  const elemento = document.getElementById(elementId);

  const canvas = await html2canvas(elemento, {
    scale: 3,          // Alta resolução para mobile
    useCORS: true,
    backgroundColor: null,
    logging: false
  });

  // Download
  const link = document.createElement('a');
  link.download = `${nomeArquivo}.png`;
  link.href = canvas.toDataURL('image/png');
  link.click();
}

// Feed 1:1 — width: 320px, aspectRatio: 1
// Stories 9:16 — width: 320px, height: 568px (padding aumentado)
```

## PARTE 7 — LÓGICA DE GERAÇÃO AUTOMÁTICA

```javascript
// Regra: cada ação gera um card específico — o usuário não escolhe

const TRIGGER_CARD = {
  "compartilhar_perfil":    "CardIdentidade",
  "compartilhar_tipo":      "CardFraseCombinada",
  "compartilhar_trecho":    "CardCitacao",      // já existe no app
  "compartilhar_progresso": "CardProgresso",
  "compartilhar_reacao":    "CardMinimal",
  "compartilhar_dna":       "CardDNAVisual"
};

function gerarCard(trigger, dadosUsuario) {
  const tipoCard = TRIGGER_CARD[trigger];
  const arquetipos = calcularArquetipos(dadosUsuario);

  switch (tipoCard) {
    case "CardIdentidade":
      return <CardIdentidade usuario={dadosUsuario} arquetipos={arquetipos} />;
    case "CardFraseCombinada":
      return <CardFraseCombinada usuario={dadosUsuario} arquetipos={arquetipos} />;
    // ... outros tipos
  }
}
```
